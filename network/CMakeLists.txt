   file(REAL_PATH "${hunter_prefix}/include" hunter_boost_include)

      NOT IS_DIRECTORY "${hunter_boost_include}")
    message(FATAL_ERROR "The Hunter Boost include directory escapes its installation")
   endif()
   if(component STREQUAL "algorithm")
    set(hunter_boost_sentinel "${hunter_boost_include}/boost/algorithm/string.hpp")
   else()
    set(hunter_boost_sentinel "${hunter_boost_include}/boost/outcome.hpp")
   endif()
   if(NOT EXISTS "${hunter_boost_sentinel}" OR IS_DIRECTORY "${hunter_boost_sentinel}")
    message(FATAL_ERROR "A required Hunter Boost header-only component is unavailable: ${component}")
   endif()
   file(REAL_PATH "${hunter_boost_sentinel}" hunter_boost_sentinel_real)

   if(IS_ABSOLUTE "${hunter_sentinel_relative}" OR hunter_sentinel_relative MATCHES "^\\.\\.(/|$)")
    message(FATAL_ERROR "A required Boost header escapes the Hunter installation: ${component}")
   endif()
   add_library("Boost::${component}" INTERFACE IMPORTED)
   set_target_properties("Boost::${component}" PROPERTIES
     INTERFACE_INCLUDE_DIRECTORIES "${hunter_boost_include}"
     INTERFACE_LINK_LIBRARIES Boost::boost)
  endif()

 function(nano_require_hunter_boost_location location)
  foreach(entry IN LISTS location)
   if(entry STREQUAL "" OR entry MATCHES "-NOTFOUND$" OR
      entry MATCHES "^(debug|optimized|general)$")
    continue()
   endif()
   if(TARGET "${entry}")
    get_target_property(configurations "${entry}" IMPORTED_CONFIGURATIONS)
    set(properties INTERFACE_INCLUDE_DIRECTORIES IMPORTED_LOCATION IMPORTED_IMPLIB)
    foreach(configuration IN LISTS configurations)
     string(TOUPPER "${configuration}" configuration)
     list(APPEND properties "IMPORTED_LOCATION_${configuration}" "IMPORTED_IMPLIB_${configuration}")
    endforeach()
    foreach(property IN LISTS properties)
     get_target_property(value "${entry}" "${property}")
     nano_require_hunter_boost_location("${value}")
    endforeach()
   elseif(IS_ABSOLUTE "${entry}" AND EXISTS "${entry}")
    file(REAL_PATH "${entry}" actual)
    file(RELATIVE_PATH relative "${hunter_prefix}" "${actual}")
    if(IS_ABSOLUTE "${relative}" OR relative MATCHES "^\\.\\.(/|$)")
     message(FATAL_ERROR "A Boost result escapes the libp2p Hunter installation: ${entry}")
    endif()
   else()
    message(FATAL_ERROR "Unverifiable Boost result in the libp2p Hunter selection: ${entry}")
   endif()
  endforeach()
 endfunction()
 get_cmake_property(boost_variables VARIABLES)
 foreach(variable IN LISTS boost_variables)
  if(variable MATCHES "^Boost_(INCLUDE_DIRS?|LIBRARY_DIRS?(_(DEBUG|RELEASE))?|LIBRARIES|[A-Z0-9_]+_LIBRAR(Y|IES)(_(DEBUG|RELEASE))?)$")
   nano_require_hunter_boost_location("${${variable}}")
  endif()
 endforeach()
 # Config packages need not publish FindBoost result variables. Enumerate
 # their imported targets directly; no recursive package/link-graph evaluator.
 get_property(boost_imported_targets DIRECTORY PROPERTY IMPORTED_TARGETS)
 list(APPEND boost_imported_targets Boost::headers Boost::boost p2p::p2p)
 list(REMOVE_DUPLICATES boost_imported_targets)
 foreach(target IN LISTS boost_imported_targets)
  if(NOT TARGET "${target}")
   continue()
  endif()
  if(target MATCHES "^(Boost::|boost_)")
   nano_require_hunter_boost_location("${target}")
  endif()
  get_target_property(links "${target}" INTERFACE_LINK_LIBRARIES)
  foreach(link IN LISTS links)
   string(REGEX MATCHALL "Boost::[A-Za-z_][A-Za-z0-9_.+-]*" references "${link}")
   foreach(reference IN LISTS references)
    if(NOT TARGET "${reference}")
     message(FATAL_ERROR "Unresolved Boost imported target: ${reference}")
    endif()
    nano_require_hunter_boost_location("${reference}")
   endforeach()
   if(link MATCHES "(^|/)libboost_[^/]*")
    nano_require_hunter_boost_location("${link}")
   elseif(TARGET "${link}" AND link MATCHES "^boost_")
    nano_require_hunter_boost_location("${link}")
   elseif(link MATCHES "[Bb]oost" AND NOT references)
    message(FATAL_ERROR "Unverifiable Boost link expression: ${link}")
   endif()
  endforeach()
 endforeach()
endif()
