cmake_minimum_required(VERSION 3.20)
project(nano_p2p LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0167)
 cmake_policy(SET CMP0167 OLD)
endif()
set(NANO_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.deps" CACHE PATH "Built cpp-libp2p dependencies")
if(EXISTS "${NANO_DEPS_DIR}/libp2p-build/CMakeCache.txt")
 file(STRINGS "${NANO_DEPS_DIR}/libp2p-build/CMakeCache.txt" boost_dir REGEX "^Boost_DIR:PATH=")
 string(REPLACE "Boost_DIR:PATH=" "" boost_dir "${boost_dir}")
 get_filename_component(hunter_prefix "${boost_dir}/../../.." ABSOLUTE)
 list(PREPEND CMAKE_PREFIX_PATH "${NANO_DEPS_DIR}/install" "${hunter_prefix}")
 # Hunter uses architecture-tagged Boost archives. CMake 3.28 does not infer
 # the tag for Linux Clang/aarch64, so retain the actual target architecture.
 if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND NOT DEFINED Boost_ARCHITECTURE)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
   set(Boost_ARCHITECTURE "-a64")
  elseif(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
   set(Boost_ARCHITECTURE "-x64")
  endif()
 endif()
 file(GLOB hunter_modules "${NANO_DEPS_DIR}/hunter/_Base/Download/Hunter/*/*/Unpacked/cmake/modules")
 list(APPEND CMAKE_MODULE_PATH ${hunter_modules})
endif()
find_package(ZLIB CONFIG REQUIRED)
find_package(qtils CONFIG REQUIRED)
find_package(libsecp256k1 CONFIG REQUIRED)
find_package(libp2p CONFIG REQUIRED)

target_link_libraries(nano-p2p PRIVATE p2p::p2p)
